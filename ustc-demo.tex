\section{开始使用}

\begin{frame}{如何使用这个模板}
  \begin{itemize}
    \item 修改导言区的 \texttt{\textbackslash title}、\texttt{\textbackslash author} 等信息。
    \item 每张幻灯片使用一个 \texttt{frame} 环境；\texttt{section} 自动插入章节目录页。
    \item 正文尽量只保留观点和支撑材料，详细解释留给口头报告。
  \end{itemize}
  \vspace{0.4cm}
  本例展示 \ctextbf{强调文字}、公式、双栏、图片、代码和结束页。
\end{frame}

\begin{frame}{列表与逐项显示}

    \item 再给出方法：用价值函数评价候选动作。

      \item 子列表可用于补充一个关键条件。
    \end{itemize}
    \item 最后说明限制：示例不代表实际实验结果。

\section{内容排版}

\begin{frame}{公式与符号}
  \begin{block}{贝尔曼最优方程（示意）}
    \[
      V^*(s)=\max_a\left[r(s,a)+\gamma\sum_{s'}P(s'\mid s,a)V^*(s')\right].
    \]
  \end{block}
  \begin{itemize}
    \item $s$ 是状态，$a$ 是动作，$\gamma\in[0,1)$ 是折扣因子。
    \item 数学公式直接使用 Beamer 自带的 \texttt{amsmath} 支持。
  \end{itemize}
\end{frame}
